📘 أوامر لينكس الأساسية

🔹 القسم الأول: 25 أمر أساسي في لينكس

1. pwd

عرض المسار الحالي للمجلد.

pwd

2. cd

التنقل بين المجلدات.

cd ..     # الرجوع للخلف
cd        # الذهاب للمجلد الرئيسي
cd -      # الرجوع للمجلد السابق

3. ls

عرض محتويات المجلد.

ls -R   # عرض الملفات داخل المجلدات الفرعية
ls -a   # عرض الملفات المخفية
ls -al  # عرض التفاصيل

4. cat

عرض محتوى الملفات أو دمجها.

cat file.txt
cat > newfile.txt
cat file1 file2 > merged.txt

5. cp

نسخ الملفات.

cp file.txt /home/user/Documents

6. mv

نقل أو إعادة تسمية الملفات.

mv file.txt /home/user/Documents
mv old.txt new.txt

7. mkdir

إنشاء مجلد جديد.

mkdir Music
mkdir -p Music/2024/New

8. rmdir

حذف مجلد فارغ.

rmdir oldfolder

9. rm

حذف الملفات أو المجلدات.

rm file.txt
rm -r folder

10. touch

إنشاء ملف فارغ.

touch index.html

11. locate

البحث عن الملفات.

locate notes.txt
locate -i school*note

12. find

البحث داخل مجلدات محددة.

find /home -name notes.txt
find . -name "*.txt"

13. grep

البحث داخل الملفات النصية.

grep blue notes.txt

14. sudo

تنفيذ الأوامر بصلاحيات المدير.

sudo apt-get update

15. df

عرض استخدام مساحة القرص.

df -m

16. du

عرض حجم الملفات والمجلدات.

du -h filename

17. head

عرض أول 10 أسطر من الملف.

head -n 5 notes.txt

18. tail

عرض آخر 10 أسطر من الملف.

tail -n 5 notes.txt

19. diff

مقارنة محتوى ملفين.

diff file1.txt file2.txt

20. tar

أرشفة الملفات.

tar -cvf archive.tar files/
tar -xvf archive.tar

21. chmod

تغيير أذونات الملفات.

chmod 755 script.sh

22. chown

تغيير مالك الملفات.

chown user:group file.txt

23. jobs

عرض العمليات النشطة في الخلفية.

jobs

24. kill

إيقاف عملية باستخدام PID.

kill 1234

25. ping

اختبار الاتصال بالخادم.

ping google.com
